add_library(libSceIcu SHARED EXCLUDE_FROM_ALL Export.cpp Utf32Conversions.cpp ModifiedUtf8.cpp StringTermination.cpp)
target_include_directories(libSceIcu PRIVATE ${LIBS_INCLUDE_DIR})
configure_windows_unwind(libSceIcu)

if(BUILD_TESTING)
    foreach(kind IN ITEMS utf32 modified_utf8 termination)
        if(kind STREQUAL "utf32")
            set(testSource GuestIcuUtf32.cpp)
        elseif(kind STREQUAL "modified_utf8")
            set(testSource GuestIcuModifiedUtf8.cpp)
        else()
            set(testSource GuestIcuTermination.cpp)
        endif()
        add_test_executable(guest_icu_${kind}_tests ${LIBS_INCLUDE_DIR}/tests/${testSource})
        target_include_directories(guest_icu_${kind}_tests PRIVATE ${LIBS_INCLUDE_DIR})
        target_link_libraries(guest_icu_${kind}_tests PRIVATE libSceIcu libc)
        configure_windows_unwind(guest_icu_${kind}_tests)
        add_test(NAME guest_icu_${kind} COMMAND guest_icu_${kind}_tests)
        set_tests_properties(guest_icu_${kind} PROPERTIES TIMEOUT 10)
        if(WIN32)
            set_tests_properties(guest_icu_${kind} PROPERTIES ENVIRONMENT_MODIFICATION
                "PATH=path_list_prepend:$<TARGET_FILE_DIR:libSceIcu>")
        endif()
    endforeach()
endif()
